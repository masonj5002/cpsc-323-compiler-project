@
    /* No Function Declarations */
@
integer i, max, sum; /* declarations */
integer x;
boolean x_is_zero;
@
    sum = 0;
    i = 1;
    read (max);
    while (i < max) {
        sum = sum + i;
        i = i + 1;
    }
    write (sum + max);

    read(x);
    
    if (x == 0)
        x_is_zero = true;
    otherwise
        x_is_zero = false;
    fi
@
